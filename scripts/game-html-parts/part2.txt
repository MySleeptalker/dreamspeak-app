let audioCtx = null;
function getCtx(){ if(!audioCtx){ audioCtx = new (window.AudioContext || window.webkitAudioContext)(); } return audioCtx; }
function playTone(freq, duration, type='sine', vol=0.18, delay=0){
  try{
    const ctx = getCtx();
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.type = type; osc.frequency.value = freq; gain.gain.value = vol;
    osc.connect(gain); gain.connect(ctx.destination);
    const startTime = ctx.currentTime + delay;
    osc.start(startTime);
    gain.gain.setValueAtTime(vol, startTime);
    gain.gain.exponentialRampToValueAtTime(0.0001, startTime + duration);
    osc.stop(startTime + duration + 0.02);
  }catch(e){}
}

let cachedVoices = [];
function loadVoices(){
  try{ cachedVoices = window.speechSynthesis.getVoices() || []; }catch(e){}
}
if('speechSynthesis' in window){
  loadVoices();
  window.speechSynthesis.onvoiceschanged = loadVoices;
}
function pickBestVoice(langCode){
  if(!cachedVoices.length) loadVoices();
  const primary = (langCode || 'en').split('-')[0].toLowerCase();
  const candidates = cachedVoices.filter(v => v.lang && v.lang.toLowerCase().startsWith(primary));
  if(!candidates.length) return null;
  const qualityHints = ['google','natural','neural','enhanced','premium','wavenet'];
  const premium = candidates.find(v => qualityHints.some(h => v.name.toLowerCase().includes(h)));
  const exact = candidates.find(v => v.lang.toLowerCase() === (langCode||'').toLowerCase());
  return premium || exact || candidates[0];
}

let currentAudioEl = null;
function stopAllAudio(){
  if(currentAudioEl){ try{ currentAudioEl.pause(); }catch(e){} currentAudioEl = null; }
  try{ if('speechSynthesis' in window) window.speechSynthesis.cancel(); }catch(e){}
}

  document.querySelectorAll('.sheep-badge').forEach(el=>{

/** Plays a real recorded clip if one exists for this language+index (alphabet stage),
 *  otherwise falls back to the browser's speech synthesis with the best available voice. */
function playRecordedClip(clipUrl){
  const audio = new Audio(clipUrl);
  currentAudioEl = audio;
  setSheepSpeaking(true);
  audio.onended = ()=>{ setSheepSpeaking(false); if(currentAudioEl === audio) currentAudioEl = null; };
  audio.onerror = ()=>{
    setSheepSpeaking(false);
    if(currentAudioEl === audio) currentAudioEl = null;
    toast('Could not play that clip — check your connection.');
  };
  audio.play().catch(()=>{ setSheepSpeaking(false); toast('Tap again to hear the sound.'); });
}

/** Plays a real recorded, professionally-voiced clip whenever one exists for this language/stage/text.
 *  Every alphabet, word, phrase, and sentence in every language has a pre-recorded clip, so this should
 *  always find a match — the browser speechSynthesis fallback only fires if a clip URL is unreachable. */
function playPronunciation(text, langKey, alphabetIndex){
  stopAllAudio();
  if(!audioManifestLoaded){
    loadAudioManifest().then(()=>playPronunciationResolved(text, langKey, alphabetIndex));
    return;
  }
  playPronunciationResolved(text, langKey, alphabetIndex);
}
function playPronunciationResolved(text, langKey, alphabetIndex){
  let clipUrl = null;
  if (typeof alphabetIndex === 'number' && ALPHABET_AUDIO[langKey]) {
    clipUrl = ALPHABET_AUDIO[langKey][alphabetIndex];
  }
  if (!clipUrl && TRANSLATION_AUDIO[langKey] && TRANSLATION_AUDIO[langKey][text]) {
    clipUrl = TRANSLATION_AUDIO[langKey][text];
  }
  if (clipUrl) {
    playRecordedClip(clipUrl);
    return;
  }
  speak(text, TTS_LANG[langKey]);
}

function speak(text, langCode){
  try{
    if(!('speechSynthesis' in window)) return;
    window.speechSynthesis.cancel();
    const cleaned = text.replace(/\([^)]*\)/g, '').trim();
    const utter = new SpeechSynthesisUtterance(cleaned || text);
    utter.lang = langCode || 'en-US';
    utter.rate = 0.72;
    utter.pitch = 1.0;
    const bestVoice = pickBestVoice(langCode);
    if(bestVoice) utter.voice = bestVoice;
    setSheepSpeaking(true);
    utter.onend = ()=>setSheepSpeaking(false);
    utter.onerror = ()=>setSheepSpeaking(false);
    window.speechSynthesis.speak(utter);
  }catch(e){}
}

  const email = document.getElementById('loginEmail').value.trim();

  errEl.classList.add('hidden');
  if(!email || !password){

    errEl.classList.remove('hidden');
    return;
  }
  const btn = document.getElementById('loginBtn');
  btn.disabled = true;
  try{
    const res = await fetch(api('/api/auth/login'), {
      method:'POST', credentials:'include',
      headers:{'Content-Type':'application/json'},
      body: JSON.stringify({ email, password })
    });
    const data = await res.json();
    if(!res.ok){
      errEl.textContent = data.error || 'Login failed.';
      errEl.classList.remove('hidden');
      return;

    afterAuthNavigate();

    errEl.textContent = 'Could not reach the server. Check your connection and try again.';
    errEl.classList.remove('hidden');
  }finally{
    btn.disabled = false;
  }
}

/* Periodically re-check with the server in case an admin changed hearts/plan. */
async function pullFromBackend(){
  if(!state.userId) return;
  try{
    const res = await fetch(api('/api/users/' + state.userId), { credentials:'include' });
    if(!res.ok) return;
    const data = await res.json();
    if(!data || !data.id) return;
    let changed = false;
    if(data.plan && data.plan !== state.plan){ state.plan = data.plan; changed = true; }
    if(typeof data.hearts === 'number' && data.hearts !== state.hearts){ state.hearts = data.hearts; changed = true; }
    if(typeof data.heartsMax === 'number' && data.heartsMax !== state.heartsMax){ state.heartsMax = data.heartsMax; changed = true; }
    if(changed){
      saveLocalCache();
      toast('🔄 Synced with your account');
      refreshHubStats();
      updateDashboard();
    }
  }catch(e){}
}

function checkAchievements(){
  ACHIEVEMENTS.forEach(a=>{
    if(!state.unlockedAchievements.includes(a.id) && a.check(state)){
      state.unlockedAchievements.push(a.id);
      sfxAchievement();
      spawnConfetti(30);
      toast(`🏅 Milestone unlocked: ${a.title}!`);
    }
  });
  saveLocalCache();
  pushToBackend();
}

function openAchievements(){
  sfxWhoosh();
  const grid = document.getElementById('achvGrid');
  grid.innerHTML = '';
  ACHIEVEMENTS.forEach(a=>{
    const unlocked = state.unlockedAchievements.includes(a.id);
    const card = document.createElement('div');
    card.className = 'achv-card glass rounded-2xl p-4 text-center border-2 border-white/10' + (unlocked ? '' : ' locked');
    card.innerHTML = `<div class="text-4xl mb-2">${a.icon}</div><div class="font-display font-bold text-sm">${a.title}</div><div class="text-xs text-white/60 mt-1">${a.desc}</div>${unlocked ? '<div class="text-[10px] mt-2 text-emerald-300 font-bold">UNLOCKED</div>' : '<div class="text-[10px] mt-2 text-white/40 font-bold">LOCKED</div>'}`;
    grid.appendChild(card);
  });
  document.getElementById('achvModal').classList.remove('hidden');
  document.getElementById('achvModal').classList.add('flex');
}
function closeAchievements(){
  document.getElementById('achvModal').classList.add('hidden');
  document.getElementById('achvModal').classList.remove('flex');
}

function regenHeartsIfDue(){
  if(state.plan === 'paid') return;
  if(state.hearts >= state.heartsMax) return;
  if(!state.lastHeartLostAt) return;
  const elapsed = Date.now() - state.lastHeartLostAt;
  if(elapsed >= HEART_REGEN_MS){
    state.hearts = state.heartsMax;
    state.lastHeartLostAt = null;
    saveLocalCache();
    toast('❤️ Your hearts have refilled!');
  }
}
function loseHeart(){
  if(state.plan === 'paid') return;
  state.hearts = Math.max(0, state.hearts - 1);
  if(!state.lastHeartLostAt){ state.lastHeartLostAt = Date.now(); }
  saveLocalCache();
  pushToBackend();
}
function heartsDisplay(){
  return state.plan === 'paid' ? '∞' : state.hearts;
}
